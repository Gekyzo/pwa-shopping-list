# Recipes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `Recetas` section where the user creates, reads, updates and deletes recipes, and sends the products of a recipe to `Actual` with one tap.

**Architecture:** The state gets a `recipes` array and becomes version 2; `src/storage.js` migrates version 1 on load and on import. A new pure module `src/recipes.js` holds the recipe operations and resolves product names against the items through `store.js`. `src/app.js` gets a fourth view that renders from the state as the other views do.

**Tech Stack:** Plain HTML, CSS and ES modules. `node:test` with `node:assert/strict`. No build step, no dependencies.

**Spec:** `docs/superpowers/specs/2026-10-12-recipes-design.md`

## Global Constraints

- Text on the screen is in Spanish. Code, comments, test names, docs and commits are in English.
- No dependency and no build step. All URLs are relative.
- An operation is a pure function `(state, …) → { state }` or `{ state, error }`. It never changes its input.
- User text goes to the DOM through `textContent` or a property, never `innerHTML`.
- The `localStorage` key stays `shopping.v1`. The `version` field becomes `2`.
- Names are trimmed. The text fields have `maxLength` 80.
- A recipe holds product names, never item ids.
- Commits: plain imperative English subject, a body for a behaviour change, no attribution line.
- Each task commit also stages `plan-current.sh task N land`.

## Review Focus

1. A product whose name differs from an item only in case (`huevos` / `Huevos`): the send reuses the item and creates no duplicate. Test in Task 2.
2. A second send of the same recipe: no item changes, `added` is 0 and the state object is the same, so the screen offers no `Deshacer`. Test in Task 2.
3. A product that is a ticked recurrent item: the send makes it not ticked and it stays recurrent. Test in Task 2.
4. A version 2 file without `recipes`, or with a recipe that has duplicate products: the import rejects it with the usual message and does not throw. Test in Task 1.
5. A product name of spaces only: `addRecipeProduct` rejects it and the recipe does not change. Test in Task 2.

```js
    id !== '' &&

  // Version 1 has no recipes.
  const rawRecipes = data.version === 1 ? [] : data.recipes;
```

```js
import { addItem, bringToList, findByName } from './store.js';

const NOT_FOUND = 'Receta no encontrada.';

const reject = (state, error) => ({ state, error });

const findRecipe = (state, id) => state.recipes.find((recipe) => recipe.id === id);

const patchRecipe = (state, id, patch) => ({
  state: { ...state, recipes: state.recipes.map((recipe) => (recipe.id === id ? { ...recipe, ...patch } : recipe)) },
});

const nameError = (state, name, ownId) => {
  if (!name) return 'Escribe un nombre.';
  const key = name.toLowerCase();
  const duplicate = state.recipes.find((recipe) => recipe.id !== ownId && recipe.name.toLowerCase() === key);
  return duplicate ? `La receta "${duplicate.name}" ya existe.` : null;
};

export const addRecipe = (state, rawName, newId = () => crypto.randomUUID()) => {
  const name = String(rawName ?? '').trim();
  const error = nameError(state, name);
  if (error) return reject(state, error);
  const recipe = { id: newId(), name, products: [] };
  return { state: { ...state, recipes: [...state.recipes, recipe] }, id: recipe.id };
};

export const renameRecipe = (state, id, rawName) => {
  if (!findRecipe(state, id)) return reject(state, NOT_FOUND);
  const name = String(rawName ?? '').trim();
  const error = nameError(state, name, id);
  return error ? reject(state, error) : patchRecipe(state, id, { name });
};

export const deleteRecipe = (state, id) => {
  if (!findRecipe(state, id)) return reject(state, NOT_FOUND);
  return { state: { ...state, recipes: state.recipes.filter((recipe) => recipe.id !== id) } };
};

export const addRecipeProduct = (state, id, rawName) => {
  const recipe = findRecipe(state, id);
  if (!recipe) return reject(state, NOT_FOUND);
  const name = String(rawName ?? '').trim();
  if (!name) return reject(state, 'Escribe un producto.');
  const key = name.toLowerCase();
  const duplicate = recipe.products.find((product) => product.toLowerCase() === key);
  if (duplicate) return reject(state, `"${duplicate}" ya está en la receta.`);
  return patchRecipe(state, id, { products: [...recipe.products, name] });
};

export const removeRecipeProduct = (state, id, name) => {
  const recipe = findRecipe(state, id);
  if (!recipe) return reject(state, NOT_FOUND);
  if (!recipe.products.includes(name)) return reject(state, 'Producto no encontrado.');
  return patchRecipe(state, id, { products: recipe.products.filter((product) => product !== name) });
};

// Puts each product of the recipe in Active as "to buy". `added` counts the items
// that were created or changed, `already` the items that were there and pending.
export const sendRecipe = (state, id, newId) => {
  const recipe = findRecipe(state, id);
  if (!recipe) return reject(state, NOT_FOUND);
  if (!recipe.products.length) return reject(state, 'La receta no tiene productos.');
  let next = state;
  let added = 0;
  for (const product of recipe.products) {
    const existing = findByName(next, product);
    const result = existing ? bringToList(next, existing.id, 'active') : addItem(next, product, 'active', newId);
    if (result.error) return reject(state, result.error);
    if (!existing || result.changed) added += 1;
    next = result.state;
  }
  return { state: next, added, already: recipe.products.length - added };
};
```

```js
  if (added) parts.push(plural(added, 'añadido', 'añadidos'));
  if (already) parts.push(`${already} ya ${already === 1 ? 'estaba' : 'estaban'}`);
  return parts.join(', ');
};

const sendToList = (recipe) => {
  const result = recipes.sendRecipe(state, recipe.id);
  if (result.error || result.added) apply(result, { undoText: `«${recipe.name}»: ${sendText(result)}.` });
  else showToast(`«${recipe.name}»: todo estaba ya en la lista.`);
};

const productNode = (recipe, product) => {
  const remove = el('button', {

    textContent: '✕',
    onclick: () =>

  });
  remove.setAttribute('aria-label', `Quitar ${product}`);
  return el('li', {}, [el('span', { textContent: product }), remove]);
};

const productForm = (recipe) => {
  const input = el('input', {
    type: 'text',
    className: 'product-name',
    maxLength: 80,
    placeholder: 'Añadir producto',
    enterKeyHint: 'done',
  });
  input.setAttribute('list', 'item-names');
  input.setAttribute('autocapitalize', 'sentences');
  input.setAttribute('aria-label', `Añadir producto a ${recipe.name}`);
  const form = el('form', { className: 'product-form', autocomplete: 'off' }, [
    input,
    el('button', { type: 'submit', className: 'primary', textContent: 'Añadir' }),
  ]);
  form.addEventListener('submit', (event) => {
    event.preventDefault();
    // The render builds a new field; `focusProduct` keeps the keyboard open for the next product.
    ui.focusProduct = true;
    if (apply(recipes.addRecipeProduct(state, recipe.id, input.value))) return;
    ui.focusProduct = false;
    input.focus();
  });
  return form;
};

const recipeNode = (recipe) => {
  const open = ui.openRecipeId === recipe.id;
  let name;
  if (ui.editingId === recipe.id) name = renameInput(recipe, (value) => recipes.renameRecipe(state, recipe.id, value));
  else {
    name = el(
      'button',
      {
        type: 'button',
        className: 'name',
        onclick: () => {
          Object.assign(ui, { openRecipeId: open ? null : recipe.id, openId: null });
          render();
        },
      },
      [
        el('span', { textContent: recipe.name }),
        el('small', { textContent: plural(recipe.products.length, 'producto', 'productos') }),
      ],
    );
    name.setAttribute('aria-expanded', String(open));
  }
  const send = el('button', { type: 'button', className: 'pill send', textContent: 'A la lista', onclick: () => sendToList(recipe) });
  send.setAttribute('aria-label', `Enviar a la lista: ${recipe.name}`);
  const node = el('li', { className: 'item recipe' }, [el('div', { className: 'row' }, [name, send, moreButton(recipe)])]);
  if (ui.openId === recipe.id) {
    const remove = actionButton(
      'Eliminar',
      () => apply(recipes.deleteRecipe(state, recipe.id), { undoText: `Receta «${recipe.name}» eliminada.` }),
      'danger',
    );
    node.append(el('div', { className: 'actions' }, [renameAction(recipe), remove]));
  }
  if (open) {
    node.append(
      el('ul', { className: 'products' }, recipe.products.map((product) => productNode(recipe, product))),
      productForm(recipe),
    );
  }
  return node;
};
```
